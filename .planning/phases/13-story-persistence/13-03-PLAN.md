    **2. Create `src/components/shared-reading-view.tsx`:**

    A `'use client'` component that receives story data from the server component and renders it in the reading view format (same as ReadingView's complete state).

    ```typescript
    'use client'

    import { useEffect, useState } from 'react'
    import { useWakeLock } from '@/lib/use-wake-lock'

    interface SharedReadingViewProps {
      story: string  // full story text, paragraphs joined with \n\n
      name: string
      theme: string
    }

    export function SharedReadingView({ story, name, theme }: SharedReadingViewProps) {
      useWakeLock()
      const [progress, setProgress] = useState(0)
      const paragraphs = story.split('\n\n').filter((p: string) => p.trim())

      useEffect(() => {
        function handleScroll() {
          const scrollTop = window.scrollY
          const docHeight = document.documentElement.scrollHeight - window.innerHeight
          setProgress(docHeight > 0 ? scrollTop / docHeight : 0)
        }
        window.addEventListener('scroll', handleScroll, { passive: true })
        return () => window.removeEventListener('scroll', handleScroll)
      }, [])

      return (
        <div className="min-h-screen bg-reading-surface">
          {/* Scroll progress bar */}
          <div
            aria-hidden="true"
            className="fixed top-0 left-0 h-[3px] z-40 transition-opacity duration-300 ease-in-out"
            style={{
              width: `${progress * 100}%`,
              backgroundColor: 'var(--color-secondary-container)',
              opacity: progress > 0 ? 1 : 0,
            }}
          />
          <div className="mx-auto max-w-[640px] px-lg pt-2xl pb-3xl">
            <article
              role="article"
              style={{
                textRendering: 'optimizeLegibility',
                WebkitFontSmoothing: 'antialiased',
              }}
            >
              <h1 className="font-serif text-[2rem] font-normal leading-[1.3] tracking-[-0.02em] text-reading-on-surface-muted mb-xl">
                {name}&apos;s {theme} Story
              </h1>
              {paragraphs.map((paragraph, index) => (
                <p
                  key={index}
                  className={`font-serif text-[1.25rem] font-normal leading-[1.8] tracking-[0.01em] text-reading-on-surface${
                    index < paragraphs.length - 1 ? ' mb-md' : ''
                  }`}
                >
                  {paragraph}
                </p>
              ))}
            </article>
            <div className="mt-2xl flex justify-center">
              <a
                href="/"
                className="bg-secondary-container text-on-secondary-container font-sans text-[0.75rem] font-semibold uppercase tracking-[0.04em] h-[48px] px-xl rounded-full inline-flex items-center justify-center transition-all duration-300 ease-in-out hover:brightness-110 focus-visible:outline-2 focus-visible:outline-reading-on-surface-muted focus-visible:outline-offset-2"
              >
                CREATE A STORY
              </a>
            </div>
          </div>
        </div>
      )
    }
    ```

    This mirrors ReadingView's complete rendering but without streaming, auto-save, or library buttons. The CTA is CREATE A STORY (not NEW STORY) since the recipient may not have used the app before.

    **3. Create `src/app/story/[shareId]/page.tsx` as a server component:**

    ```typescript
    import type { Viewport } from 'next'
    import { getRedis } from '@/lib/redis'
    import { ExpiredStoryView } from '@/components/expired-story-view'
    import { SharedReadingView } from '@/components/shared-reading-view'

    export const viewport: Viewport = {
      themeColor: '#1a1a2e',
    }

    export default async function SharedStoryPage({
      params,
    }: {
      params: Promise<{ shareId: string }>
    }) {
      const { shareId } = await params  // Next.js 15+ async params per Pitfall 5

      const redis = getRedis()
      if (!redis) {
        return <ExpiredStoryView />
      }

      const raw = await redis.get<string>(`share:${shareId}`)
      if (!raw) {
        return <ExpiredStoryView />
      }

      const data = typeof raw === 'string' ? JSON.parse(raw) : raw
      return <SharedReadingView story={data.story} name={data.name} theme={data.theme} />
    }
    ```

    Per D-10: `/story/[shareId]` renders full ReadingView. Async `await params` per Next.js 15+ (Pitfall 5). No Edge runtime (anti-pattern from RESEARCH.md). Server-side fetch means no loading spinner for recipients.

    Per D-11: when Redis returns null (expired or not found), render ExpiredStoryView — NOT `notFound()`.
