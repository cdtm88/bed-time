    async function handleShare(e: React.MouseEvent) {
      e.stopPropagation() // Don't trigger re-read
      setShareLoading(true)
      try {
        const res = await fetch('/api/share', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ story: story.story, name: story.name, theme: story.theme }),
        })
        const { shareId } = await res.json()
        const url = `${window.location.origin}/story/${shareId}`
        await navigator.clipboard.writeText(url)
        setCopied(true)
        setTimeout(() => setCopied(false), 2000)
      } catch {
        // Silent failure — could show error but keeping simple per discretion
      } finally {
        setShareLoading(false)
      }
    }
